function doGet() { return HtmlService.createHtmlOutputFromFile("index") .setTitle("AUA Steel Fabrication Attendance") .addMetaTag("viewport", "width=device-width, initial-scale=1"); } function getEmployees() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheetByName("Employee_Masterlist"); if (!sheet) { throw new Error("Employee_Masterlist sheet was not found."); } const data = sheet.getDataRange().getDisplayValues(); if (data.length < 2) { return []; } const headers = data[0].map(header => String(header).trim().toLowerCase().replace(/\s+/g, " ") ); const nameColumn = headers.indexOf("employee name"); const emailColumn = headers.indexOf("company email"); const statusColumn = headers.indexOf("status"); if ( nameColumn === -1 || emailColumn === -1 || statusColumn === -1 ) { throw new Error( "Required headers: Employee Name, Company Email, Status." ); } return data.slice(1) .filter(row => String(row[nameColumn]).trim() !== "" && String(row[statusColumn]).trim().toLowerCase() === "active" ) .map(row => ({ name: String(row[nameColumn]).trim(), email: String(row[emailColumn]).trim() })); } function doPost(e) { try { if (!e || !e.parameter) { throw new Error("No attendance submission was received."); } const employeeName = String( e.parameter.employeeName || "" ).trim(); Logger.log("Submitted employee name: [" + employeeName + "]"); Logger.log("Active masterlist names: " + JSON.stringify(getEmployees())); const attendanceType = String( e.parameter.attendanceType || "" ).trim(); const photo = String(e.parameter.photo || ""); if (!employeeName || !attendanceType || !photo) { throw new Error( "Employee name, attendance type, and photo are required." ); } if (!["Time-In", "Time-Out"].includes(attendanceType)) { throw new Error("Invalid attendance type."); } // Retrieve active employees from the masterlist const employees = getEmployees(); // Normalize names to ignore capitalization and extra spaces const normalizeName = name => String(name) .normalize("NFKC") .replace(/\s+/g, " ") .trim() .toLowerCase(); // Match the submitted employee against the active masterlist const employee = employees.find(item => normalizeName(item.name) === normalizeName(employeeName) ); if (!employee) { const activeNames = employees.map(item => item.name).join(", "); throw new Error( "Employee not found or inactive: " + employeeName + ". Active employees: " + activeNames ); } if ( !employee.email || employee.email.toLowerCase() === "pending" ) { throw new Error( "The employee's company email is missing." ); } // Validate and decode the captured photo const photoParts = photo.split(","); if (photoParts.length < 2) { throw new Error("Invalid photo data."); } const photoBlob = Utilities.newBlob( Utilities.base64Decode(photoParts[1]), "image/jpeg", employee.name.replace(/[^\w\s-]/g, "") + "_" + attendanceType + "_" + new Date().getTime() + ".jpg" ); // Find or create the attendance photo folder const folderName = "AUA Attendance Photos"; const folders = DriveApp.getFoldersByName(folderName); const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); // Save the attendance photo const photoFile = folder.createFile(photoBlob); const photoUrl = photoFile.getUrl(); // Get the attendance spreadsheet const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const attendanceSheet = spreadsheet.getSheetByName("Attendance_Log"); if (!attendanceSheet) { throw new Error( "Attendance_Log sheet was not found." ); } // Get the current date and time const now = new Date(); const timezone = Session.getScriptTimeZone(); const attendanceDate = Utilities.formatDate( now, timezone, "yyyy-MM-dd" ); // Append the attendance record attendanceSheet.appendRow([ now, employee.name, employee.email, attendanceType, attendanceDate, photoUrl, "Pending Verification", "" ]); return HtmlService.createHtmlOutput( "

Attendance recorded successfully.

" + "

Your attendance has been submitted.

" + "

You may close this tab and return to the attendance page.

Attendance submission failed.
